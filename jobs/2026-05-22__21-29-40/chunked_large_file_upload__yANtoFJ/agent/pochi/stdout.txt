I have implemented the Node.js script to upload a large file to Appwrite Storage with chunked progress reporting.

The script performs the following actions:
1.  **Configures the Appwrite Client** using `APPWRITE_ENDPOINT`, `APPWRITE_PROJECT_ID`, and `APPWRITE_API_KEY`.
2.  **Creates a new Storage bucket** with ID and name `bkt-$ZEALT_RUN_ID`.
3.  **Uploads the 6.5 MiB file** located at `/home/user/myproject/sample.bin` using `InputFile.fromPath`.
4.  **Reports progress to stderr** as JSON objects for each chunk, including `chunksTotal`, `chunksUploaded`, `progress`, `sizeUploaded`, and `id`.
5.  **Grants public read permission** to the uploaded file.
6.  **Outputs the resulting file ID** to stdout.

The implementation uses the `node-appwrite` SDK and follows the specific requirements for progress reporting and output formatting.

```javascript
// /home/user/myproject/index.js
const { Client, Storage, ID, Permission, Role } = require('node-appwrite');
const { InputFile } = require('node-appwrite/file');

async function run() {
    const endpoint = process.env.APPWRITE_ENDPOINT;
    const project = process.env.APPWRITE_PROJECT_ID;
    const key = process.env.APPWRITE_API_KEY;
    const runId = process.env.ZEALT_RUN_ID;

    if (!endpoint || !project || !key || !runId) {
        process.exit(1);
    }

    const client = new Client()
        .setEndpoint(endpoint)
        .setProject(project)
        .setKey(key);

    const storage = new Storage(client);
    const bucketId = `bkt-${runId}`;
    const filePath = '/home/user/myproject/sample.bin';

    try {
        await storage.createBucket(bucketId, bucketId);

        const file = await storage.createFile(
            bucketId,
            ID.unique(),
            InputFile.fromPath(filePath, 'sample.bin'),
            [Permission.read(Role.any())],
            (progress) => {
                const log = {
                    chunksTotal: progress.chunksTotal,
                    chunksUploaded: progress.chunksUploaded,
                    progress: progress.progress,
                    sizeUploaded: progress.sizeUploaded,
                    id: progress.$id
                };
                process.stderr.write(JSON.stringify(log) + '\n');
            }
        );

        console.log(file.$id);
    } catch (error) {
        process.exit(1);
    }
}

run();
```   
