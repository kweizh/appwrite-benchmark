const { Client, Databases } = require("node-appwrite");
async function cleanup() {
    const client = new Client().setEndpoint(process.env.APPWRITE_ENDPOINT).setProject(process.env.APPWRITE_PROJECT_ID).setKey(process.env.APPWRITE_API_KEY);
    const databases = new Databases(client);
    const list = await databases.list();
    console.log(`Found ${list.total} databases`);
    for (const db of list.databases) {
        if (db.name.startsWith("tasks_db_")) {
            console.log(`Deleting database: ${db.$id} (${db.name})`);
            await databases.delete(db.$id);
        }
    }
}
cleanup().catch(console.error);
